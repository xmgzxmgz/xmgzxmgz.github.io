🎨 色彩空间转换

OpenCV 默认使用 BGR 格式，和常见的 RGB 顺序不同。转换到不同色彩空间可以看到图像的不同特征：

import cv2
import numpy as np

img = cv2.imread('photo.jpg')

# BGR → RGB（用于 matplotlib 显示）
rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)

# BGR → 灰度
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# BGR → HSV（色调-饱和度-明度）
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

# BGR → YUV（亮度-色度）
yuv = cv2.cvtColor(img, cv2.COLOR_BGR2YUV)

HSV 空间特别有用 —— 在 HSV 空间中做颜色筛选比 RGB 容易得多。

📐 仿射变换

仿射变换可以实现平移、旋转、缩放和剪切：

# 旋转 45 度
h, w = img.shape[:2]
center = (w//2, h//2)
M = cv2.getRotationMatrix2D(center, 45, 1.0)
rotated = cv2.warpAffine(img, M, (w, h))

# 平移
M = np.float32([[1, 0, 100], [0, 1, 50]])
translated = cv2.warpAffine(img, M, (w, h))

🔍 边缘检测

Canny 边缘检测是最常用的算法，通过双阈值控制边缘的粗细：

# Canny 边缘检测
edges = cv2.Canny(gray, threshold1=50, threshold2=150)

# Robert 算子（手动实现）
kernel_x = np.array([[1, 0], [0, -1]])
kernel_y = np.array([[0, 1], [-1, 0]])
robert_x = cv2.filter2D(gray, -1, kernel_x)
robert_y = cv2.filter2D(gray, -1, kernel_y)
robert = cv2.addWeighted(robert_x, 0.5, robert_y, 0.5, 0)

🧹 噪声与滤波

给图像添加椒盐噪声，然后用不同滤波器去除：

# 添加椒盐噪声
def add_salt_pepper(img, prob=0.02):
    noisy = img.copy()
    num_salt = int(prob * img.size / 2)
    # 盐噪声（白点）
    coords = [np.random.randint(0, i, num_salt) for i in img.shape[:2]]
    noisy[coords[0], coords[1]] = 255
    # 椒噪声（黑点）
    coords = [np.random.randint(0, i, num_salt) for i in img.shape[:2]]
    noisy[coords[0], coords[1]] = 0
    return noisy

# 均值滤波
mean_filtered = cv2.blur(noisy, (5, 5))

# 中值滤波（对椒盐噪声效果最好）
median_filtered = cv2.medianBlur(noisy, 5)
